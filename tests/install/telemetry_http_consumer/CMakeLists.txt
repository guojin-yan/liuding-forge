cmake_minimum_required(VERSION 3.18 FATAL_ERROR)
project(jyppx_telemetry_http_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(jyppx CONFIG REQUIRED COMPONENTS telemetry_http)
if(NOT jyppx_telemetry_http_FOUND OR NOT TARGET jyppx::liuding::telemetry_http)
    message(FATAL_ERROR "The installed package must supply telemetry_http")
endif()
if(TARGET jyppx::liuding::sdk)
    set(expected)
    foreach(name IN ITEMS foundation core interop data nn optim training experiment)
        list(APPEND expected jyppx::liuding::${name})
    endforeach()
    get_target_property(actual jyppx::liuding::sdk INTERFACE_LINK_LIBRARIES)
    list(SORT expected)
    list(SORT actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "HTTP must not enter the eight-Core SDK aggregate")
    endif()
endif()
add_executable(telemetry_http_consumer main.cpp)
target_link_libraries(telemetry_http_consumer PRIVATE jyppx::liuding::telemetry_http)
if(WIN32)
    if(NOT JYPPX_TORCH_RUNTIME_DIR OR NOT EXISTS "${JYPPX_RUNTIME_STAGER}")
        message(FATAL_ERROR "Windows HTTP consumer requires the Torch runtime and StageTorchRuntime.cmake")
    endif()
    add_custom_command(TARGET telemetry_http_consumer POST_BUILD
        COMMAND ${CMAKE_COMMAND}
            "-DJYPPX_RUNTIME_SOURCE_DIR=${JYPPX_TORCH_RUNTIME_DIR}"
            "-DJYPPX_RUNTIME_OUTPUT_DIR=$<TARGET_FILE_DIR:telemetry_http_consumer>"
            -P "${JYPPX_RUNTIME_STAGER}" VERBATIM)
endif()
enable_testing()
add_test(NAME Runs COMMAND telemetry_http_consumer)
set_tests_properties(Runs PROPERTIES TIMEOUT 30)
