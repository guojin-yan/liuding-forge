cmake_minimum_required(VERSION 3.18 FATAL_ERROR)
project(jyppx_migraphx_serving_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
if(JYPPX_MIGRAPHX_SERVING_ONLY)
    find_package(jyppx CONFIG REQUIRED COMPONENTS serving)
    if(TARGET jyppx_external_migraphx_c OR TARGET jyppx_external_migraphx_hip)
        message(FATAL_ERROR "Serving-only consumer discovered unused MIGraphX SDK")
    endif()
    add_executable(consumer main.cpp)
    target_compile_definitions(consumer PRIVATE JYPPX_MIGRAPHX_SERVING_ONLY=1)
    target_link_libraries(consumer PRIVATE jyppx::liuding::serving)
else()
    find_package(jyppx CONFIG REQUIRED COMPONENTS migraphx_serving)
    add_executable(consumer main.cpp)
    target_link_libraries(consumer PRIVATE jyppx::liuding::migraphx_serving)
endif()
enable_testing()
add_test(NAME Runs COMMAND consumer "${JYPPX_MIGRAPHX_CONSUMER_FIXTURES}")
set_tests_properties(Runs PROPERTIES TIMEOUT 600)
