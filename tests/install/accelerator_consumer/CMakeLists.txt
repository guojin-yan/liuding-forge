cmake_minimum_required(VERSION 3.18 FATAL_ERROR)
project(jyppx_accelerator_consumer LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
    add_compile_definitions(_ITERATOR_DEBUG_LEVEL=0)
endif()
find_package(jyppx CONFIG REQUIRED COMPONENTS sdk)
add_executable(jyppx_accelerator_consumer main.cpp)
target_link_libraries(jyppx_accelerator_consumer PRIVATE jyppx::liuding::sdk)
enable_testing()
add_test(NAME RunsOnAccelerator COMMAND jyppx_accelerator_consumer)
