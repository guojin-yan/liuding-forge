cmake_minimum_required(VERSION 3.18)
project(liuding_build_provenance_fixture VERSION 1.0.0 LANGUAGES CXX)

if(NOT JYPPX_PROVENANCE_MODULE OR NOT TEST_BUILD_COMMIT OR NOT TEST_BUILD_LOCK)
    message(FATAL_ERROR "Build provenance fixture inputs are required")
endif()
set(JYPPX_GIT_COMMIT "${TEST_BUILD_COMMIT}")
set(JYPPX_DEPENDENCY_LOCK_DIGEST "${TEST_BUILD_LOCK}")
configure_file(identity.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/identity.hpp" @ONLY)
add_library(provenance_fixture STATIC identity.cpp)
target_include_directories(provenance_fixture PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
include("${JYPPX_PROVENANCE_MODULE}")
jyppx_install_build_provenance(provenance_fixture)
install(TARGETS provenance_fixture ARCHIVE DESTINATION lib)
add_executable(provenance_probe probe.cpp)
target_link_libraries(provenance_probe PRIVATE provenance_fixture)
get_property(is_multi GLOBAL PROPERTY GENERATOR_IS_MULTI_CONFIG)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/multi-config.txt" "${is_multi}")
